  - name : Implement SSh Jey Authenttcation
    hosts : all
    become : yes
    tasks :
       - name : Generate SSh Key pair
         command :  ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N ""
       - name :  Add SSH public key to authoirized_keys
         authorized_key :
	    user : user
	    state : present
	    Key : ={{lookup('file', '/home/user/.ssh/id_rsa.pub')}}'
